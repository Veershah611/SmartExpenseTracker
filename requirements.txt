# --- UI / Frontend ---------------------------------------------------------
streamlit>=1.40.0            # Dashboard, tabs, chat UI, session state
# Required by backend/analytics_engine.py (Analytics role). The shell still

plotly>=5.24.0

# --- Data Processing -------------------------------------------------------
pandas>=2.2.0                # All aggregation / group-by / rolling analytics
numpy>=1.26.0                # Vector math + fallback cosine-similarity search

# --- Database --------------------------------------------------------------
# sqlite3 ships with the Python standard library -> no dependency needed.

# --- Vector Database (semantic search over expenses & advice) --------------
# Chroma is the primary store. If it fails to build on Python 3.13, comment it

# --- LLM Integration (local, offline-friendly) -----------------------------

requests>=2.32.0             # Ollama + LM Studio HTTP clients, health checks

# --- Vision / OCR (receipt scanning) ---------------------------------------
opencv-python-headless>=4.10.0   # Pre-processing: greyscale, threshold, deskew
pillow>=10.4.0                   # Image I/O bridge between Streamlit and cv2
pytesseract>=0.3.13              # Tesseract bindings
# NOTE: pytesseract is only the Python binding. The Tesseract BINARY is a
# separate install and must be on PATH, or receipt scanning fails at runtime:
#   Windows: https://github.com/UB-Mannheim/tesseract/wiki
#   macOS:   brew install tesseract
#   Linux:   apt install tesseract-ocr

# --- Config / Utilities ----------------------------------------------------
python-dotenv>=1.0.1         # Loads .env so no secrets/hosts are hard-coded
